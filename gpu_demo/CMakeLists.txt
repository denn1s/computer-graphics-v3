cmake_minimum_required(VERSION 3.16)
project(shader_workshop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 REQUIRED)
find_package(glm REQUIRED)
add_executable(shader_workshop main.cpp)
target_link_libraries(shader_workshop PRIVATE OpenGL::GL GLEW::GLEW glfw glm::glm)
target_compile_options(shader_workshop PRIVATE -Wall -Wextra -Wpedantic)
